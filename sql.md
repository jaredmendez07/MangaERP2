create table public.perfiles (
  id uuid primary key references auth.users(id) on delete cascade,
  nombre text not null,
  correo text not null,
  rol text not null check (rol in ('admin', 'user')),
  estado text not null default 'activo',
  fecha_registro timestamp with time zone default now()
);

create table public.materia_prima (
  id bigint generated by default as identity primary key,
  nombre text not null,
  categoria text not null,
  cantidad_disponible numeric not null default 0,
  unidad text not null,
  stock_minimo numeric not null default 0,
  estado text not null default 'disponible',
  created_at timestamp with time zone default now()
);

create table public.productos_manga (
  id bigint generated by default as identity primary key,
  titulo text not null,
  volumen int not null,
  tipo_edicion text not null check (tipo_edicion in ('Estándar', 'Especial', 'Coleccionista')),
  paginas int not null,
  estado text not null default 'activo',
  created_at timestamp with time zone default now()
);

create table public.produccion (
  id bigint generated by default as identity primary key,
  fecha date not null,
  producto_id bigint references public.productos_manga(id) on delete set null,
  cantidad_producida int not null,
  linea_impresion text not null,
  estado_produccion text not null check (estado_produccion in ('Pendiente', 'En impresión', 'Encuadernado', 'Empaquetado', 'Finalizado')),
  operador text,
  created_at timestamp with time zone default now()
);

create table public.consultas_ia (
  id bigint generated by default as identity primary key,
  usuario_id uuid references auth.users(id) on delete set null,
  pregunta text not null,
  respuesta text,
  tipo_consulta text,
  created_at timestamp with time zone default now()
);
